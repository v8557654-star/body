ИНСТРУКЦИЯ ДЛЯ URP в 2022.3.76f1 LTS

Твоя версия 2022.3.76f1 Расширенная LTS идет с URP 14.0.11 по дефолту.

У тебя 2 варианта:

ВАРИАНТ 1 (Рекомендую для старта) - Built-in RP:
1. При создании проекта выбери 3D (Built-in), НЕ URP
2. Тогда шейдер OceanShader.shader заработает сразу
3. Поставь его на материал океана

ВАРИАНТ 2 - URP (красивее):
1. Создай проект 3D (URP)
2. Создай Shader Graph -> Unlit или PBR
3. Воспроизведи логику из OceanShader.shader:

В Shader Graph:
- Создай Custom Function Node с кодом Gerstner (скопируй из OceanManager.cs SampleWaves)
- 4 волны достаточно для вершин
- Position Node -> Add Gerstner displacement -> Vertex Position
- Для цвета: lerp DeepColor/ShallowColor по waveHeight + Fresnel
- Добавь пена: step(waveHeight, 1.2)

Или используй готовый URP Shader код ниже - создай файл OceanURP.shader:

------------------------------------------------------------
Shader "SeaDrift/OceanURP"
{
    Properties
    {
        _DeepColor ("Deep", Color) = (0.02,0.15,0.3,1)
        _ShallowColor ("Shallow", Color) = (0.05,0.5,0.6,1)
        _Smoothness ("Smoothness", Range(0,1)) = 0.9
    }
    SubShader
    {
        Tags { "RenderType"="Opaque" "RenderPipeline"="UniversalPipeline" }
        Pass
        {
            Name "ForwardLit"
            Tags { "LightMode"="UniversalForward" }
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Lighting.hlsl"

            struct Attributes { float4 positionOS : POSITION; float3 normalOS : NORMAL; float2 uv : TEXCOORD0; };
            struct Varyings { float4 positionCS : SV_POSITION; float3 positionWS : TEXCOORD0; float3 normalWS : TEXCOORD1; float waveHeight : TEXCOORD2; };

            CBUFFER_START(UnityPerMaterial)
                float4 _DeepColor; float4 _ShallowColor; float _Smoothness;
            CBUFFER_END

            float3 Gerstner(float2 dir, float steep, float len, float amp, float speed, float3 pos, float t, inout float3 tan, inout float3 bin)
            {
                float k = 6.283185 / len;
                float c = sqrt(9.8 / k);
                float f = k * (dot(dir, pos.xz) - c * t * speed * 0.1);
                float s = sin(f); float co = cos(f);
                float3 disp;
                disp.x = -dir.x * amp * s * steep;
                disp.y = amp * co;
                disp.z = -dir.y * amp * s * steep;
                tan.x += -k * dir.x * dir.x * steep * amp * co;
                tan.y += -k * dir.x * amp * s;
                tan.z += -k * dir.x * dir.y * steep * amp * co;
                bin.x += -k * dir.y * dir.x * steep * amp * co;
                bin.y += -k * dir.y * amp * s;
                bin.z += -k * dir.y * dir.y * steep * amp * co;
                return disp;
            }

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                float3 pos = IN.positionOS.xyz;
                float3 tan = float3(1,0,0); float3 bin = float3(0,0,1); float3 disp=0;
                float t = _Time.y;
                float3 worldPos = TransformObjectToWorld(pos);
                disp += Gerstner(normalize(float2(1,0.3)),0.12,60,1.5,8, worldPos, t, tan, bin);
                disp += Gerstner(normalize(float2(0.7,0.7)),0.08,35,0.8,6, worldPos, t, tan, bin);
                disp += Gerstner(normalize(float2(-0.3,1)),0.05,20,0.4,4, worldPos, t, tan, bin);
                disp += Gerstner(normalize(float2(0.2,-1)),0.04,12,0.25,3, worldPos, t, tan, bin);
                pos += disp;
                OUT.positionWS = TransformObjectToWorld(pos);
                OUT.normalWS = normalize(cross(bin, tan));
                OUT.waveHeight = disp.y;
                OUT.positionCS = TransformWorldToHClip(OUT.positionWS);
                return OUT;
            }

            half4 frag(Varyings IN) : SV_Target
            {
                float3 viewDir = normalize(_WorldSpaceCameraPos - IN.positionWS);
                float fresnel = pow(1 - saturate(dot(IN.normalWS, viewDir)), 5);
                float hf = saturate(IN.waveHeight*0.3+0.5);
                float3 col = lerp(_DeepColor.rgb, _ShallowColor.rgb, hf + fresnel*0.3);
                float foam = saturate(IN.waveHeight-0.8)*2;
                col = lerp(col, 1, foam);
                Light mainLight = GetMainLight();
                float diff = saturate(dot(IN.normalWS, mainLight.direction));
                col *= diff*0.5+0.5;
                col += pow(saturate(dot(reflect(-mainLight.direction, IN.normalWS), viewDir)), 80)*0.8;
                return half4(col,1);
            }
            ENDHLSL
        }
    }
}
------------------------------------------------------------

Просто создай новый Shader в Unity, вставь этот код, сохрани.

Для 2022.3.76f1 все скрипты совместимы на 100% - я использую только базовый Rigidbody и Mathf, без новых API.

Если проект просит апгрейд при открытии - жми Upgrade, все ок.
